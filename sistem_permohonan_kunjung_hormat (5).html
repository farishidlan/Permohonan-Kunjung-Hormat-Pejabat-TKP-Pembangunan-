Konfigurasi Sambungan Google Sheets

Panduan untuk Setiausaha Pejabat (SUP) mengaktifkan Google Excel automatik.

Bagaimana ia berfungsi?

Setiap kali orang awam atau kontraktor isi borang di web ini, maklumat akan dihantar serta-merta ke fail Google Sheets (Excel) milik pejabat melalui Google Apps Script Webhook.

*URL ini disimpan dalam sistem pelayar anda untuk penghantaran terus.

Langkah Setup Simpan PDF ke Google Drive & Sheets:
  1. Buka Google Sheets pejabat anda.
  2. Di menu atas, klik Extensions → Apps Script.
  3. Gantikan kod lama dengan kod baharu di bawah (ia akan automatik simpan PDF ke Google Drive dan jana link di Sheets):
function doPost(e) {
  try {
    var sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet();
    var data = JSON.parse(e.postData.contents);
    
    // 1. Simpan dokumen PDF ke Google Drive pejabat secara automatik
    var fileUrl = "-";
    var hyperlinkFormula = "-";
    
    if (data.fileBase64 && data.fileName) {
      var folderName = "Surat Permohonan Kunjung Hormat TKP";
      var folders = DriveApp.getFoldersByName(folderName);
      var folder = folders.hasNext() ? folders.next() : DriveApp.createFolder(folderName);
      
      var decoded = Utilities.base64Decode(data.fileBase64);
      var blob = Utilities.newBlob(decoded, data.fileType || "application/pdf", data.fileName);
      var file = folder.createFile(blob);
      
      // Berikan akses bacaan kepada sesiapa yang ada link
      file.setSharing(DriveApp.Access.ANYONE_WITH_LINK, DriveApp.Permission.VIEW);
      fileUrl = file.getUrl();
      hyperlinkFormula = '=HYPERLINK("' + fileUrl + '", "📄 Buka Surat PDF")';
    }
    
    // 2. Jika baris pertama kosong, tambah tajuk lajur
    if (sheet.getLastRow() === 0) {
      sheet.appendRow([
        "Timestamp", "No Rujukan", "Nama Syarikat", 
        "Nama Wakil", "Jawatan Wakil", "No Telefon", 
        "Emel", "Cadangan Tarikh", "Cadangan Masa", 
        "Bil Delegasi", "Tujuan Kunjungan", "Nama Fail Surat", "Pautan Google Drive (Surat)", "Status"
      ]);
    }
    
    // 3. Tambah data permohonan baharu bersama Link Google Drive
    sheet.appendRow([
      new Date().toLocaleString("ms-MY"),
      data.referenceNo,
      data.companyName,
      data.representativeName,
      data.representativeDesignation,
      data.phone,
      data.email,
      data.proposedDate,
      data.proposedTime,
      data.delegateCount,
      data.purpose,
      data.fileName,
      hyperlinkFormula !== "-" ? hyperlinkFormula : fileUrl,
      "Baharu"
    ]);
    
    return ContentService.createTextOutput(JSON.stringify({status: "success", fileUrl: fileUrl}))
      .setMimeType(ContentService.MimeType.JSON);
  } catch (err) {
    return ContentService.createTextOutput(JSON.stringify({status: "error", message: err.toString()}))
      .setMimeType(ContentService.MimeType.JSON);
  }
}
  1. Klik Deploy (kanan atas) → Manage Deployments → Edit (ikon pensel) → Pilih Version New version → Klik Deploy.
  2. Kini apabila pemohon menghantar borang, fail PDF akan masuk ke folder Google Drive dan lajur ke-13 di Google Sheet akan bertukar menjadi butang pautan biru "📄 Buka Surat PDF"!
